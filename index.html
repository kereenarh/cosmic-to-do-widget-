  <script>

    // Asset references for confetti

    const CONFETTI_ASSETS = [

      "confetti_stars.png",

      "confetti_mix.png",

      "confetti_moons.png"

    ];


    const DELETE_BUTTON_SRC = "delete_button.png";


    // Selectors

    const form = document.getElementById("todo-form");

    const input = document.getElementById("task-input");

    const list = document.getElementById("todo-list");

    const confettiLayer = document.getElementById("confetti-layer");


    // Load saved tasks from LocalStorage

    let tasks = JSON.parse(localStorage.getItem("cloud_tasks") || "[]");


    function saveTasks() {

      localStorage.setItem("cloud_tasks", JSON.stringify(tasks));

    }


    // Render list

    function renderTasks() {

      list.innerHTML = "";

      tasks.forEach((task, index) => {

        const li = document.createElement("li");

        li.className = `todo-item ${task.completed ? "completed" : ""}`;


        // Left section (Star + Task Name)

        const leftDiv = document.createElement("div");

        leftDiv.className = "todo-left";

        leftDiv.onclick = () => toggleTask(index);


        const star = document.createElement("span");

        star.className = "star-checkbox";

        star.innerHTML = task.completed ? "★" : "☆";


        const text = document.createElement("span");

        text.className = "todo-text";

        text.textContent = task.text;


        leftDiv.appendChild(star);

        leftDiv.appendChild(text);


        // Delete button

        const delBtn = document.createElement("button");

        delBtn.className = "icon-btn";

        delBtn.title = "Delete item";

        delBtn.onclick = (e) => {

          e.stopPropagation();

          deleteTask(index);

        };


        const delImg = document.createElement("img");

        delImg.src = DELETE_BUTTON_SRC;

        delImg.alt = "x";

        delBtn.appendChild(delImg);


        li.appendChild(leftDiv);

        li.appendChild(delBtn);

        list.appendChild(li);

      });

    }


    // Add Task

    form.addEventListener("submit", (e) => {

      e.preventDefault();

      const val = input.value.trim();

      if (!val) return;


      tasks.push({ text: val, completed: false });

      input.value = "";

      saveTasks();

      renderTasks();

    });


    // Toggle Task completion

    function toggleTask(index) {

      const isNowCompleted = !tasks[index].completed;

      tasks[index].completed = isNowCompleted;

      saveTasks();

      renderTasks();


      if (isNowCompleted) {

        triggerConfetti();

      }

    }


    // Delete Task

    function deleteTask(index) {

      tasks.splice(index, 1);

      saveTasks();

      renderTasks();

    }


    // Celebrate task completion with 5-second confetti

    function triggerConfetti() {

      const pieceCount = 16;


      for (let i = 0; i < pieceCount; i++) {

        const piece = document.createElement("img");

        // Pick one of the confetti assets randomly

        piece.src = CONFETTI_ASSETS[Math.floor(Math.random() * CONFETTI_ASSETS.length)];

        piece.className = "confetti-piece";


        // Random horizontal position across widget

        piece.style.left = `${Math.random() * 80 + 10}%`;

        // Random starting vertical spread near bottom

        piece.style.bottom = `${Math.random() * 40 + 20}px`;

        // Random delay to stagger the float

        piece.style.animationDelay = `${Math.random() * 0.4}s`;


        confettiLayer.appendChild(piece);


        // Automatically clean up each piece after 5 seconds

        setTimeout(() => {

          piece.remove();

        }, 5000);

      }

    }


    // Initial render

    renderTasks();

  </script>
